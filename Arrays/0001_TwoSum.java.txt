import java.util.*;

class Solution {
    public int[] twoSum(int[] nums, int target) {
        HashMap<Integer, Integer> map = new HashMap<>(); //we are creating a hashmap variable here
        
        for (int i = 0; i < nums.length; i++) {
            int needed = target - nums[i];  
            if (map.containsKey(needed)) {
                return new int[] { map.get(needed), i };//create a list and return the i and the stored needed from the past iterations 
            }   
            map.put(nums[i], i);
        }
        return new int[] {};//returning empty if nothing exists 
    }
}
